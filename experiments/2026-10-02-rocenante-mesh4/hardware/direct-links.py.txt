import subprocess,shlex,json,time
from pathlib import Path
root=Path('/Users/christopherowen/projects/spark3-kernel-64k')
nodes=json.loads((root/'config/nodes-ring4.local.json').read_text())['nodes'];by={n['rank']:n for n in nodes}
out=root/'.work/mesh4-qualification/run01-direct';out.mkdir(parents=True,exist_ok=True)
results=[]
def ssh(n,cmd):return ['ssh','-o','BatchMode=yes','-o','ConnectTimeout=8','swank@'+n['name'],shlex.join(cmd)]
for rank in range(4):
 peer=(rank+1)%4;n=by[rank];p=by[peer]
 for lane in range(2):
  h=n['roce_peer_hcas'][str(peer)][lane];ph=p['roce_peer_hcas'][str(rank)][lane]
  ip=p['mesh_ports'][ph]['address'].split('/')[0];dev=n['mesh_ports'][h]['netdev']
  label=f'{n["name"]}-{p["name"]}-lane{lane}'
  ping=subprocess.run(ssh(n,['ping','-I',dev,'-M','do','-s','8972','-c','2','-W','1',ip]),capture_output=True,text=True,timeout=8)
  (out/(label+'-ping.txt')).write_text(ping.stdout+ping.stderr)
  port=18600+rank*2+lane
  common=['timeout','25s','ib_write_bw','-x','3','-F','--report_gbits','-b','-s','65536','-n','1000','-p',str(port)]
  server_cmd=ssh(n,common+['-d',h]);client_cmd=ssh(p,common+['-d',ph,n['management_ip']])
  server=subprocess.Popen(server_cmd,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
  time.sleep(.5)
  client=subprocess.run(client_cmd,capture_output=True,text=True,timeout=30)
  server_out=server.communicate(timeout=30)[0]
  (out/(label+'-server.txt')).write_text(server_out);(out/(label+'-client.txt')).write_text(client.stdout+client.stderr)
  r={'case':label,'ping_exit':ping.returncode,'server_exit':server.returncode,'client_exit':client.returncode,'server_command':server_cmd,'client_command':client_cmd}
  results.append(r);(out/'results.json').write_text(json.dumps(results,indent=2)+'\n');print(json.dumps(r),flush=True)
