# Queue: bash B/bin/run_all.sh pr-957p1k pr-957p1s pr-957p1n pr-957p2 pr-957p4 pr-988 pr-957p1t pr-957p1v pr-957p4d

# #957 part 1 (0795)

## B/bin/pr-957p1k.sh
# #957 part 1 (0795): the compressor projection cells (h957k.py), ABBA, then the PR's test.
B=$HOME/spark-lab/agents/round3-own-marks; OM=$B/bin/om.sh; . $B/bin/env.sh; mkdir -p $B/out/pr-957p1/json
for spec in "base 1" "c56p1 1" "c56p1 2" "base 2"; do set -- $spec
  $OM 957p1 proj-$1-boot$2 $1 "${RECIPE_ENV[@]}" -- python3 /h/h957k.py --boot $2 --out /out/json/proj-$1-boot$2.json
done

## B/bin/pr-957p1s.sh
# #957 part 1 (0795): the decode step at TP4 rank shapes on one GPU (step957.py), ABBA.
B=$HOME/spark-lab/agents/round3-own-marks; OM=$B/bin/om.sh; . $B/bin/env.sh; mkdir -p $B/out/pr-957p1/json
for spec in "base 1" "c56p1 1" "c56p1 2" "base 2"; do set -- $spec
  MINMEM_GIB=95 $OM 957p1 step-$1-boot$2 $1 "${RECIPE_ENV[@]}" "${HUB[@]}" -- python3 /h/step957.py --out /out/json/step-$1-boot$2.json
done

## B/bin/pr-957p1n.sh
# #957 part 1 (0795): one nsys boot an arm (kernel timeline of the grid's 1x2 and 1x6 steps,
# graph nodes traced), after the timing boots.
B=$HOME/spark-lab/agents/round3-own-marks; OM=$B/bin/om.sh; . $B/bin/env.sh; mkdir -p $B/out/pr-957p1/json $B/out/pr-957p1/nsys
for arm in base c56p1; do
  MINMEM_GIB=95 $OM 957p1 nsys-$arm $arm "${RECIPE_ENV[@]}" "${HUB[@]}" "${NSYS[@]}" -- /opt/nsys/bin/nsys profile \
    --trace=cuda --sample=none --cpuctxsw=none --cuda-graph-trace=node --capture-range=cudaProfilerApi \
    --capture-range-end=stop --export=sqlite --force-overwrite=true --output /out/nsys/nsys-$arm \
    python3 /h/step957.py --out /out/json/nsys-$arm.json --shapes 1x2,1x6,16x32 --replays 3 \
    --profile-shapes 1x2,1x6 --profile-replays 4
done

## B/bin/pr-957p1t.sh
# #957 part 1 (0795): tests/models/test_deepseek_v4_1_compressor.py, as candidate-56's README runs it.
B=$HOME/spark-lab/agents/round3-own-marks; OM=$B/bin/om.sh
for spec in "c56p1 1" "c56p1 2" "base 1"; do set -- $spec
  $OM 957p1 pytest-$1-boot$2 $1 -- pytest --noconftest -o cache_dir=/tmp/pc -q tests/models/test_deepseek_v4_1_compressor.py
done

## B/bin/pr-957p1v.sh
# #957 part 1 (0795): mechanism controls, one variable each: the L2 prefetch off
# (VLLM_DS41_L2_PREFETCH=0), and only the COMP window dropped (--no-comp-window).
B=$HOME/spark-lab/agents/round3-own-marks; OM=$B/bin/om.sh; . $B/bin/env.sh
S=1x2,1x3,1x6,4x8,16x32
for spec in "base 1" "c56p1 1" "c56p1 2" "base 2"; do set -- $spec
  MINMEM_GIB=95 $OM 957p1 nopf-$1-boot$2 $1 "${RECIPE_ENV[@]}" -e VLLM_DS41_L2_PREFETCH=0 "${HUB[@]}" -- python3 /h/step957.py --shapes $S --out /out/json/nopf-$1-boot$2.json
done
for spec in "base 1" "c56p1 1" "c56p1 2" "base 2"; do set -- $spec
  MINMEM_GIB=95 $OM 957p1 nocomp-$1-boot$2 $1 "${RECIPE_ENV[@]}" "${HUB[@]}" -- python3 /h/step957.py --shapes $S --no-comp-window --out /out/json/nocomp-$1-boot$2.json
done
